    import javax.swing.*;
    import java.awt.*;
    import java.awt.event.*;

        JFrame frame = new JFrame("Licznik");

            new zad().run();

            frame.setVisible(true);
        }
    }

```javascript tangle:helloWorld.js
    package lab02;

    import java.awt.FlowLayout;

    import javax.swing.*;
    import java.awt.*;
    import java.awt.event.*;

    public class zad implements ActionListener {

        JFrame frame = new JFrame("Licznik");
        JButton button = new JButton("Kliknij!");
        JLabel label = new JLabel("0");
        
        JComboBox<Counter> counters = new JComboBox<>(new Counter[]{
                new SimpleCounter(), 
                new DoubleCounter(),
        });

        @Override
        public void actionPerformed(ActionEvent e) {
            Counter counter = (Counter) counters.getSelectedItem();
            counter.increase();
            label.setText(Integer.toString(counter.currentValue()));
            frame.pack();
        }
        
        public static void main(String[] args) {
            new zad().run();
        }

        void run() {
            frame.setLayout(new FlowLayout());
            frame.add(button);
            frame.add(label);
            frame.add(counters);

            button.addActionListener(this);

            frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
            frame.pack();
            frame.setVisible(true);
        }
    }
```
